## Prerequisites (do first)

### CC1. `sync-plugin` script

The cache-vs-dev divergence problem (discovered 2026-05-03) means edits to plugin source don't propagate to the loaded copy without explicit sync. Every plugin edit during Phase 0 will need this script.

Write `~/.claude/bin/sync-plugin`:
```bash
#!/bin/bash
# sync-plugin <plugin-name> — copy plugin from dev to cache so next session loads edits
set -euo pipefail
PLUGIN="${1:?Usage: sync-plugin <plugin-name>}"
DEV="$HOME/.claude/plugins/$PLUGIN"
CACHE_BASE="$HOME/.claude/plugins/cache/fearsidhe-plugins/$PLUGIN"
[ -d "$DEV" ] || { echo "ERROR: dev path $DEV not found"; exit 1; }
[ -d "$CACHE_BASE" ] || { echo "ERROR: cache base $CACHE_BASE not found"; exit 1; }
LATEST=$(ls "$CACHE_BASE" | sort -V | tail -1)
CACHE="$CACHE_BASE/$LATEST"
echo "Syncing $DEV → $CACHE"
rsync -av --delete "$DEV/" "$CACHE/"
echo "Done. Next session will load updated plugin."
```

Make executable: `chmod +x ~/.claude/bin/sync-plugin`. Verify: `which sync-plugin` returns the path.

### Step 5: Write `lib/config.py`

```python
"""Config loader with fallback from user override to plugin defaults."""
from __future__ import annotations

import os
from pathlib import Path
from typing import Any

import yaml

_USER_CONFIG = Path.home() / ".config" / "continuity" / "config.yaml"
_PLUGIN_DEFAULTS = Path(__file__).parent.parent / "config.example.yaml"


def load_config() -> dict[str, Any]:
    """Load config: user override merged over plugin defaults.

    User config wins per-key. Missing user file → defaults only.
    """
    with _PLUGIN_DEFAULTS.open() as f:
        config = yaml.safe_load(f)

    if _USER_CONFIG.exists():
        with _USER_CONFIG.open() as f:
            user = yaml.safe_load(f) or {}
        config = _deep_merge(config, user)

    config["_loaded_from"] = str(_USER_CONFIG) if _USER_CONFIG.exists() else "defaults-only"
    return config


def _deep_merge(base: dict, overlay: dict) -> dict:
    result = dict(base)
    for k, v in overlay.items():
        if k in result and isinstance(result[k], dict) and isinstance(v, dict):
            result[k] = _deep_merge(result[k], v)
        else:
            result[k] = v
    return result


def vault_path(config: dict, *parts: str) -> Path:
    """Build an absolute path under vault_root."""
    return Path(config["vault_root"]).expanduser().joinpath(*parts)
```

Save as `~/.claude/plugins/continuity/lib/config.py`.

```python
#!/usr/bin/env python3
"""continuity — CLI entry point. Subcommands stubbed in Phase 0."""
from __future__ import annotations

import argparse
import sys
from pathlib import Path

# Make plugin lib/ importable
sys.path.insert(0, str(Path(__file__).parent.parent))

from lib.config import load_config


def cmd_index(args):

def cmd_recent(args):

    return 0


def cmd_backlinks(args):
    print(f"[stub] continuity backlinks {args.path} — find referrers (Phase 2)")
    return 0


def cmd_bootstrap(args):
    print(f"[stub] continuity bootstrap {args.path or '<cwd>'} — create project scaffolding (Phase 3)")
    return 0


def cmd_recap(args):
    print("[stub] continuity recap — daily recap with promotion candidates (Phase 5)")
    return 0


def cmd_promote(args):
    print(f"[stub] continuity promote {args.source} --target {args.target} — manual promotion (Phase 5)")
    return 0


def cmd_sync(args):
    print("[stub] continuity sync — vault git pull/push (Phase 6)")
    return 0


def cmd_install(args):
    """One-time setup. Phase 0: validates config + reports vault paths."""
    config = load_config()
    vault = Path(config["vault_root"]).expanduser()
    print(f"continuity install — config loaded from {config['_loaded_from']}")
    print(f"  vault_root:        {vault} {'(exists)' if vault.exists() else '(MISSING)'}")
    print(f"  continuity_tenant: {vault / config['continuity_tenant']}")
    print(f"  index_path:        {vault / config['index_path']}")
    print(f"  expose_mcp:        {config['expose_mcp']}")
    print(f"  promotion_engine:  {config['recap_promotion_engine']}")
    if not vault.exists():
        print("ERROR: vault_root does not exist. Edit ~/.config/continuity/config.yaml")
        return 1
    return 0


def main(argv=None):
    parser = argparse.ArgumentParser(prog="continuity", description="Vault-native lazy-load memory")
    sub = parser.add_subparsers(dest="cmd", required=True)

    sub.add_parser("index", help="Print the lazy-load index").set_defaults(func=cmd_index)

    p_cap = sub.add_parser("capture", help="Record an observation")
    p_cap.add_argument("--kind", required=True, choices=["observation", "decision", "general", "inbox"])
    p_cap.add_argument("--project")
    p_cap.add_argument("--slug")
    p_cap.add_argument("body", nargs="?", default="")
    p_cap.set_defaults(func=cmd_capture)

    p_search = sub.add_parser("search", help="Search the vault")
    p_search.add_argument("query")
    p_search.set_defaults(func=cmd_search)

    p_read = sub.add_parser("read", help="Read a vault file")
    p_read.add_argument("path")
    p_read.set_defaults(func=cmd_read)

    p_recent = sub.add_parser("recent", help="List recent changes")
    p_recent.add_argument("--days", type=int, default=7)
    p_recent.set_defaults(func=cmd_recent)

    p_back = sub.add_parser("backlinks", help="Find files referencing target")
    p_back.add_argument("path")
    p_back.set_defaults(func=cmd_backlinks)

    p_boot = sub.add_parser("bootstrap", help="Create a new project")
    p_boot.add_argument("path", nargs="?")
    p_boot.add_argument("--name")
    p_boot.set_defaults(func=cmd_bootstrap)

    sub.add_parser("recap", help="Daily recap with promotion candidates").set_defaults(func=cmd_recap)

    p_prom = sub.add_parser("promote", help="Manual promotion")
    p_prom.add_argument("source")
    p_prom.add_argument("--target", required=True)
    p_prom.set_defaults(func=cmd_promote)

    sub.add_parser("sync", help="Vault git pull/push").set_defaults(func=cmd_sync)
    sub.add_parser("install", help="One-time setup").set_defaults(func=cmd_install)

    args = parser.parse_args(argv)
    return args.func(args)


if __name__ == "__main__":
    sys.exit(main())
```

## Risks during Phase 0

- **Hook syntax error breaks Claude session loading.** Mitigation: test `hooks/hooks.json` against a known-working example (agent-swarm's hooks.json). Validate JSON with `jq` before committing.
- **Plugin registration format wrong.** Mitigation: examine `installed_plugins.json` entries for already-working plugins; mirror the format exactly.
- **Vault `.gitattributes` change breaks existing files.** Mitigation: union merge is additive; existing files are unaffected. Test by `git status` before and after the change.
- **WSL filesystem atomicity issues** (per CC2). Mitigation: if test fails, document and plan staging-then-mv pattern for Phase 1.

## What Phase 0 explicitly does NOT do

- Read or write any vault content beyond creating the empty `_continuity/` directory tree
- Inject any context at SessionStart
- Provide any actual capture, search, read, sync, or recap functionality
- Migrate any existing memory files
- Affect agent-swarm or any other plugin

These are all subsequent phases.
